## 二、采集器接口

```typescript
/**
 * 采集器统一接口
 * 所有采集器实现此接口，返回统一格式的 RawEvidence
 */
interface Collector {
  readonly type: AccessMethod;

  /**
   * 执行采集
   * @param source - 目标情报源
   * @param seeds  - 本次使用的种子
   * @returns 采集到的原始证据列表
   */
  collect(source: IntelSource, seeds: Seed[]): Promise<RawEvidence[]>;

  /**
   * 健康检查（可选实现）
   * @returns 采集器是否可用
   */
  healthCheck(): Promise<boolean>;
}

/**
 * 采集器工厂
 */
class CollectorFactory {
  private collectors = new Map<AccessMethod, Collector>();

  register(collector: Collector): void {
    this.collectors.set(collector.type, collector);
  }

  getCollector(method: AccessMethod): Collector {
    const collector = this.collectors.get(method);
    if (!collector) {
      throw new Error(`No collector registered for method: ${method}`);
    }
    return collector;
  }
}
```

### 3.1 RSS 采集器

> 最稳定、最快的采集方式。优先使用。

```typescript
/**
 * RSS 采集器
 * 使用 rss-parser 库
 * 
 * 适用源类型：
 * - RSSHub 生成的 RSS（TapTap、微博、B站等）
 * - WeWe RSS 生成的公众号 RSS
 * - 原生 RSS 源（GameLook、游戏葡萄等）
 */
class RssCollector implements Collector {
  readonly type = 'rss';

  async collect(source: IntelSource, seeds: Seed[]): Promise<RawEvidence[]> {
    const feed = await this.parseFeed(source.feedUrl || source.baseUrl);
    
    // 用种子过滤：标题/描述包含种子关键词
    const filtered = this.filterBySeeds(feed.items, seeds);
    
    return filtered.map(item => this.toEvidence(item, source, seeds));
  }

  private filterBySeeds(items: FeedItem[], seeds: Seed[]): FeedItem[] {
    const keywords = seeds.map(s => s.text.toLowerCase());
    const aliasKeywords = seeds
      .filter((s): s is EntitySeed => s.seedType === 'entity')
      .flatMap(s => s.aliases.map(a => a.toLowerCase()));
    
    const allKeywords = [...keywords, ...aliasKeywords];
    
    // 如果种子列表为空，则不过滤（全量采集）
    if (allKeywords.length === 0) return items;
    
    return items.filter(item => {
      const text = `${item.title} ${item.contentSnippet || ''}`.toLowerCase();
      return allKeywords.some(kw => text.includes(kw));
    });
  }

  private toEvidence(item: FeedItem, source: IntelSource, seeds: Seed[]): RawEvidence {
    return {
      id: generateId(),
      sourceId: source.id,
      seedIds: seeds.map(s => s.id),
      url: normalizeUrl(item.link),
      title: item.title || '',
      content: item.contentSnippet || item.content || '',
      contentHtml: item.content || undefined,
      author: item.creator || item.author,
      publishedAt: item.isoDate || undefined,
      collectedAt: new Date().toISOString(),
      images: extractImages(item.content || ''),
      metadata: {
        collectorType: 'rss',
        guid: item.guid,
        categories: item.categories,
      },
      hash: computeSimHash(item.title + (item.contentSnippet || '')),
      status: 'collected',
    };
  }
}
```

### 3.2 API 采集器

> 对接官方/第三方 API，数据最结构化。

```typescript
/**
 * API 采集器
 * 支持 REST API、GraphQL 等
 * 
 * 适用源类型：
 * - Steam API（新游/更新/热门）
 * - App Store / Google Play（榜单/评分）
 * - 版号查询 API
 * - 第三方数据 API
 */
class ApiCollector implements Collector {
  readonly type = 'api';

  async collect(source: IntelSource, seeds: Seed[]): Promise<RawEvidence[]> {
    const config = source.config;
    const response = await this.fetchApi(source.baseUrl, config);
    return this.parseResponse(response, source, seeds);
  }

  private async fetchApi(url: string, config: SourceConfig): Promise<any> {
    const headers = {
      'User-Agent': config.userAgent || 'GI-Bot/1.0',
      ...config.headers,
    };

    const response = await fetch(url, {
      headers,
      signal: AbortSignal.timeout(30_000),
    });

    if (!response.ok) {
      throw new CollectorError(`API error: ${response.status}`, url);
    }

    return response.json();
  }
}
```

### 3.3 Crawl4AI 采集器

> 静态页采集主力。通过 Python sidecar 调用。

```typescript
/**
 * Crawl4AI 采集器
 * 调用 Python sidecar 进行静态页采集
 * 
 * 特点：
 * - 中文优化
 * - 内置 OCR（Tesseract）
 * - 自动提取正文
 * - 支持图片 OCR
 */
class Crawl4AICollector implements Collector {
  readonly type = 'static_crawl';
  private sidecarUrl: string;  // Python sidecar 地址

  async collect(source: IntelSource, seeds: Seed[]): Promise<RawEvidence[]> {
    // 1. 获取要爬取的页面 URL 列表
    const urls = await this.buildCrawlList(source, seeds);

    // 2. 批量发送到 Python sidecar
    const results = await this.callSidecar(urls, {
      ocr: true,            // 启用 OCR
      extractImages: true,  // 提取图片
      maxPages: source.config.maxPages || 10,
      cssSelectors: source.config.cssSelectors,
    });

    // 3. 转为 RawEvidence
    return results.map(r => this.toEvidence(r, source, seeds));
  }

  /**
   * 调用 Python sidecar
   */
  private async callSidecar(urls: string[], options: CrawlOptions): Promise<CrawlResult[]> {
    const response = await fetch(`${this.sidecarUrl}/crawl`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ urls, options }),
    });

    return response.json();
  }
}

// Python sidecar 接口定义
interface CrawlRequest {
  urls: string[];
  options: {
    ocr: boolean;
    extractImages: boolean;
    maxPages: number;
    cssSelectors?: Record<string, string>;
  };
}

interface CrawlResult {
  url: string;
  title: string;
  content: string;          // 提取的纯文本
  contentHtml: string;      // 原始 HTML
  summary: string;
  images: {
    url: string;
    alt: string;
    ocrText: string;        // OCR 识别结果
    ocrConfidence: number;
  }[];
  metadata: Record<string, string>;
  error?: string;
}
```

### 3.4 Playwright 采集器

> 动态页采集。需要浏览器渲染的页面。

```typescript
/**
 * Playwright 采集器
 * 使用 Playwright 进行动态页采集
 * 
 * 适用场景：
 * - 需要 JavaScript 渲染的页面
 * - 需要登录/交互的页面
 * - SPA 应用
 * 
 * 注意：
 * - 资源消耗大，控制并发
 * - 需要设置合理的超时和重试
 */
class PlaywrightCollector implements Collector {
  readonly type = 'dynamic';
  private browser: Browser | null = null;

  async collect(source: IntelSource, seeds: Seed[]): Promise<RawEvidence[]> {
    const page = await this.getPage();
    
    try {
      // 1. 导航到目标页面
      await page.goto(source.baseUrl, {
        waitUntil: 'networkidle',
        timeout: 60_000,
      });

      // 2. 如果需要登录
      if (source.config.requiresAuth) {
        await this.handleAuth(page, source);
      }

      // 3. 等待内容加载
      if (source.config.cssSelectors?.content) {
        await page.waitForSelector(source.config.cssSelectors.content, {
          timeout: 30_000,
        });
      }

      // 4. 提取内容
      const content = await page.evaluate((selectors) => {
        // 在浏览器上下文中执行
        const extractFromSelectors = (sels: Record<string, string>) => {
          // ...
        };
        return extractFromSelectors(selectors);
      }, source.config.cssSelectors || {});

      // 5. 滚动加载更多（如需要）
      if (source.config.depth && source.config.depth > 1) {
        await this.scrollAndCollect(page, source.config.depth);
      }

      return this.toEvidenceArray(content, source, seeds);
    } finally {
      // 不关闭页面，复用 browser
    }
  }

  private async getPage(): Promise<Page> {
    if (!this.browser) {
      this.browser = await chromium.launch({ headless: true });
    }
    return this.browser.newPage();
  }
}
```

**并发控制**：
```yaml

  max_concurrent_pages: 3     # 最多同时 3 个页面
  default_timeout: 60000      # 默认超时 60s
  navigation_timeout: 30000   # 导航超时 30s
  reuse_browser: true         # 复用 browser 实例
  headless: true
```

### 3.5 Search 采集器

> 搜索引擎检索。用于定向搜索种子关键词。

```typescript
/**
 * 搜索采集器
 * 使用搜索引擎 API 检索种子关键词
 * 
 * 适用场景：
 * - 定向搜索特定实体/事件
 * - 搜索兜底（补全 RSS 覆盖不到的）
 * - 发现新源（搜索相关网站）
 * 
 * 支持的搜索引擎：
 * - Brave Search API（免费额度）
 * - SearXNG（自建，零成本）
 * - DuckDuckGo（非官方 API）
 */
class SearchCollector implements Collector {
  readonly type = 'search';

  async collect(source: IntelSource, seeds: Seed[]): Promise<RawEvidence[]> {
    const queries = this.buildQueries(seeds, source);
    const results: RawEvidence[] = [];

    for (const query of queries) {
      const searchResults = await this.search(query, source);
      const evidences = await Promise.all(
        searchResults.map(r => this.fetchAndParse(r, source, seeds))
      );
      results.push(...evidences.filter(Boolean) as RawEvidence[]);

      // 请求间隔
      await sleep(source.config.requestDelay || 2000);
    }

    return results;
  }

  private buildQueries(seeds: Seed[], source: IntelSource): string[] {
    return seeds
      .filter(s => s.seedType === 'entity' || s.seedType === 'event')
      .map(seed => {
        if (seed.seedType === 'entity') {
          return `${seed.text} 游戏 最新消息`;
        }
        if (seed.seedType === 'event') {
          return `${(seed as EventSeed).keywords.join(' ')} 游戏`;
        }
        return seed.text;
      });
  }

  private async search(query: string, source: IntelSource): Promise<SearchResult[]> {
    // 使用 SearXNG 自建实例
    const response = await fetch(`${source.baseUrl}/search`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        q: query,
        format: 'json',
        categories: 'news',
        language: 'zh-CN',
        time_range: 'week',  // 只搜最近一周
      }),
    });
    return response.json();
  }
}
```

## 四、采集路由

```typescript
/**
 * 采集路由分发器
 * 根据源的 accessMethod 自动选择最佳采集器
 */
class CollectorRouter {
  private factory: CollectorFactory;
  private health: SourceHealthService;

  /**
   * 执行一次采集任务
   */
  async route(job: CollectionJob): Promise<RawEvidence[]> {
    const source = await this.getSource(job.sourceId);
    const seeds = await this.getSeeds(job.seedIds);
    
    // 选择采集器
    const collector = this.factory.getCollector(source.accessMethod);

    // 记录任务开始
    const startedAt = new Date();

    try {
      // 执行采集
      const evidences = await collector.collect(source, seeds);
      
      // 去重（URL + SimHash）
      const newEvidences = await this.dedup(evidences);

      // 更新源健康状态
      await this.health.recordSuccess(source.id, {
        duration: Date.now() - startedAt.getTime(),
        evidenceCount: evidences.length,
        newEvidenceCount: newEvidences.length,
      });

      // 更新种子评分
      await this.updateSeedScores(seeds, newEvidences);

      return newEvidences;
    } catch (error) {
      // 记录失败
      await this.health.recordFailure(source.id, error);

      // 重试逻辑
      if (source.schedule.retryOnFail && job.retryCount < job.maxRetries) {
        const backoff = source.schedule.backoffMinutes * (job.retryCount + 1);
        await this.scheduleRetry(job, backoff);
      }

      throw error;
    }
  }

  /**
   * 去重：URL 规范化 + SimHash
   */
  private async dedup(evidences: RawEvidence[]): Promise<RawEvidence[]> {
    const result: RawEvidence[] = [];

    for (const evidence of evidences) {
      // URL 去重
      const existingUrl = await this.findByUrl(evidence.url);
      if (existingUrl) {
        evidence.status = 'duplicate';
        continue;
      }

      // SimHash 去重（标题 + 内容）
      const similar = await this.findBySimHash(evidence.hash, 0.9);
      if (similar) {
        evidence.status = 'duplicate';
        continue;
      }

      result.push(evidence);
    }

    return result;
  }
}
```

## 五、错误处理

```typescript
/**
 * 采集器错误体系
 */
class CollectorError extends Error {
  constructor(
    message: string,
    public readonly sourceId: string,
    public readonly sourceName: string,
    public readonly cause?: unknown,
  ) {
    super(`[${sourceName}] ${message}`);
    this.name = 'CollectorError';
  }
}

// 分类错误
class NetworkError extends CollectorError {}      // 网络错误
class TimeoutError extends CollectorError {}       // 超时
class AuthError extends CollectorError {}          // 认证失败
class ParseError extends CollectorError {}         // 解析失败
class RateLimitError extends CollectorError {}     // 被限流
```

**重试策略**：
```yaml
retry:
  network_error:
    max_retries: 3
    backoff: exponential    # 1s, 2s, 4s
    retry_on: [ETIMEDOUT, ECONNREFUSED, ECONNRESET]

  rate_limit:
    max_retries: 1
    backoff: fixed
    wait_minutes: 5

  auth_error:
    max_retries: 0          # 不重试，需要人工介入
    alert: true

  parse_error:
    max_retries: 0          # 不重试，需要修复选择器
    alert: true
```

## 十一、设计决策

| 决策 | 结论 | 理由 |
|------|------|------|
| Python sidecar | Crawl4AI 通过 HTTP 调用 | Node.js 无法直接调用 Python 库 |
| Playwright 并发 | 最多 3 页 | 控制资源消耗 |
| 去重在采集层 | URL + SimHash | 早期去重，减少后续处理量 |
| 搜索兜底 | 低频定期执行 | 补全 RSS 覆盖不到的 |
| 错误分类重试 | 不同错误不同策略 | 网络错误可重试，解析错误需修复 |
| 图片 OCR | 在 Crawl4AI 侧完成 | 减少 Node.js 负担 |
