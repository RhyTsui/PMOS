# 02-数据模型设计

> 版本：1.0 | 创建时间：2026-06-18 | 状态：设计中
> 本文定义 GI（游戏内参）所有核心数据结构。

---

## 一、全局数据流

```
情报源 (IntelSource)
    ↓ 采集器
原始证据 (RawEvidence) ← 种子 (Seed) 驱动
    ↓ LLM 抽取
结构化事件 (StructuredEvent)
    ↓ 去重 + 合并
证据事件 (EvidenceEvent)
    ↓ 评分
信号 (Signal)
    ↓ 聚合
趋势簇 (TrendCluster)
    ↓ 同步
Dataki 知识库
```

## 二、核心模型定义

### 2.1 情报源 IntelSource

> 一个可采集的信息入口：一个网站、一个 RSS 源、一个 API。

```typescript
interface IntelSource {
  id: string;                    // UUID
  name: string;                  // 源名称，如 "GameLook"
  shortName: string;             // 缩写，如 "GL"
  sourceType: SourceType;        // 源类型
  accessMethod: AccessMethod;    // 采集方式
  baseUrl: string;               // 入口 URL
  feedUrl?: string;              // RSS feed URL（如有）
  config: SourceConfig;          // 源特定配置
  schedule: ScheduleConfig;      // 调度配置
  enabled: boolean;              // 是否启用
  priority: Priority;            // 优先级
  tags: string[];                // 标签：['官方', '行业', '社区']
  createdAt: string;             // ISO 8601
  updatedAt: string;
}

type SourceType =
  | 'media'        // 行业媒体（GameLook、游戏葡萄）
  | 'community'    // 社区（TapTap、NGA、贴吧）
  | 'official'     // 官方渠道（Steam、App Store、版署）
  | 'social'       // 社媒（微博、B站）
  | 'wechat_mp'    // 微信公众号
  | 'forum'        // 论坛/贴吧
  | 'api';         // 数据 API

type AccessMethod =
  | 'rss'          // RSS 订阅（最快、最稳）
  | 'api'          // 官方/第三方 API
  | 'static_crawl' // Crawl4AI 静态页采集
  | 'dynamic'      // Playwright 动态页采集
  | 'search';      // 搜索引擎检索

type Priority = 'P0' | 'P1' | 'P2' | 'P3';

interface SourceConfig {
  // 采集控制
  maxPages?: number;             // 每次采集最大页数
  depth?: number;                // 爬取深度
  cssSelectors?: {               // CSS 选择器配置
    title?: string;
    content?: string;
    publishDate?: string;
    images?: string;
  };
  // 反爬策略
  requestDelay?: number;         // 请求间隔（ms）
  userAgent?: string;
  headers?: Record<string, string>;
  // 认证
  requiresAuth?: boolean;
  authMethod?: 'cookie' | 'token' | 'none';
}

interface ScheduleConfig {
  cron: string;                  // node-cron 表达式
  retryOnFail: boolean;          // 失败是否重试
  maxRetries: number;            // 最大重试次数
  backoffMinutes: number;        // 重试退避时间
}
```

**SQLite 表结构**：
```sql
CREATE TABLE intel_sources (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL UNIQUE,
  short_name TEXT NOT NULL,
  source_type TEXT NOT NULL CHECK(source_type IN ('media','community','official','social','wechat_mp','forum','api')),
  access_method TEXT NOT NULL CHECK(access_method IN ('rss','api','static_crawl','dynamic','search')),
  base_url TEXT NOT NULL,
  feed_url TEXT,
  config TEXT NOT NULL DEFAULT '{}',    -- JSON
  schedule TEXT NOT NULL DEFAULT '{}',  -- JSON
  enabled INTEGER NOT NULL DEFAULT 1,
  priority TEXT NOT NULL DEFAULT 'P1' CHECK(priority IN ('P0','P1','P2','P3')),
  tags TEXT NOT NULL DEFAULT '[]',      -- JSON array
  created_at TEXT NOT NULL DEFAULT (datetime('now')),
  updated_at TEXT NOT NULL DEFAULT (datetime('now'))
);
```

### 2.2 种子 Seed

> 种子是采集系统的发动机。四类种子驱动不同的采集策略。

```typescript
// 种子基础结构
interface BaseSeed {
  id: string;
  seedType: SeedType;
  text: string;                  // 种子文本（关键词/实体名/话题标签/URL 片段）
  score: number;                 // 0-100，种子质量分
  status: SeedStatus;
  discoveryCount: number;        // 累计发现证据数
  lastUsedAt?: string;           // 最近一次用于采集的时间
  lastEffectiveAt?: string;      // 最近一次产出有效证据的时间
  failCount: number;             // 连续失败次数
  tags: string[];                // 关联标签
  createdAt: string;
  updatedAt: string;
}

type SeedStatus =
  | 'active'       // 活跃使用中
  | 'dormant'      // 休眠（长期无效但未淘汰）
  | 'degraded'     // 降级（评分低于阈值）
  | 'retired';     // 已淘汰

// —— 四类种子 ——

/**
 * 实体种子：游戏名、公司名、人名
 * 驱动定向搜索 + 内容匹配
 */
interface EntitySeed extends BaseSeed {
  seedType: 'entity';
  entityType: 'game' | 'company' | 'person' | 'brand' | 'ip';
  aliases: string[];             // 别名（如 "王者荣耀" → "HOK"、"AoV"）
  category?: string;             // 品类：SLG / RPG / 二游 / 休闲
  market?: string;               // 市场：国内 / 出海 / 全球
}

/**
 * 事件种子：关注特定类型的事件
 * 驱动 RSS 过滤 + 搜索查询构造
 */
interface EventSeed extends BaseSeed {
  seedType: 'event';
  eventType: EventType;          // 事件类型
  keywords: string[];            // 触发关键词组合
  // 示例：{ eventType: '版号', keywords: ['版号', '审批', '国产网游'] }
}

/**
 * 话题种子：关注某个话题/趋势
 * 驱动搜索流 + 话题追踪
 */
interface TopicSeed extends BaseSeed {
  seedType: 'topic';
  topicTag: string;              // 话题标签
  relatedEntities: string[];     // 关联实体 ID
  trendDirection?: 'rising' | 'stable' | 'declining';
}

/**
 * 源种子：发现新信息源的线索
 * 驱动源发现流程
 */
interface SourceSeed extends BaseSeed {
  seedType: 'source';
  sourceType: SourceType;
  discoveryUrl?: string;         // 发现的候选 URL
  discoveryMethod: string;       // 发现方式：'link_extract' | 'search' | 'referral'
  verified: boolean;             // 是否已验证为可用源
}

// 种子联合体
type Seed = EntitySeed | EventSeed | TopicSeed | SourceSeed;
type SeedType = 'entity' | 'event' | 'topic' | 'source';

// 事件类型（情报视角）
type EventType =
  | '上线'         // 游戏上线/公测/正式服
  | '测试'         // 内测/删档测试/技术测试
  | '预约'         // 预约开启/预约量里程碑
  | '版号'         // 版号下发/审批
  | '榜单变化'     // 畅销榜/下载榜变化
  | '买量'         // 大规模投放/素材变化
  | '舆情'         // 负面舆论/玩家抗议/BUG事件
  | '融资'         // 融资/并购/IPO
  | '组织动作'     // 裁员/架构调整/高管变动
  | '版本更新'     // 大版本/新赛季/新角色
  | '出海'         // 海外发行/本地化
  | '合作'         // IP联动/品牌合作
  | '政策'         // 行业政策/监管变化
  | 'AI应用';      // AI 在游戏行业的应用
```

**SQLite 表结构**：
```sql
CREATE TABLE seeds (
  id TEXT PRIMARY KEY,
  seed_type TEXT NOT NULL CHECK(seed_type IN ('entity','event','topic','source')),
  text TEXT NOT NULL,
  score REAL NOT NULL DEFAULT 50,
  status TEXT NOT NULL DEFAULT 'active' CHECK(status IN ('active','dormant','degraded','retired')),
  -- 实体种子特有
  entity_type TEXT CHECK(entity_type IN ('game','company','person','brand','ip')),
  aliases TEXT DEFAULT '[]',               -- JSON
  category TEXT,
  market TEXT,
  -- 事件种子特有
  event_type TEXT,
  keywords TEXT DEFAULT '[]',              -- JSON
  -- 话题种子特有
  topic_tag TEXT,
  related_entities TEXT DEFAULT '[]',      -- JSON
  trend_direction TEXT CHECK(trend_direction IN ('rising','stable','declining')),
  -- 源种子特有
  discovery_url TEXT,
  discovery_method TEXT,
  verified INTEGER DEFAULT 0,
  -- 通用字段
  discovery_count INTEGER NOT NULL DEFAULT 0,
  last_used_at TEXT,
  last_effective_at TEXT,
  fail_count INTEGER NOT NULL DEFAULT 0,
  tags TEXT NOT NULL DEFAULT '[]',         -- JSON
  created_at TEXT NOT NULL DEFAULT (datetime('now')),
  updated_at TEXT NOT NULL DEFAULT (datetime('now'))
);

-- 索引：按类型 + 状态查询
CREATE INDEX idx_seeds_type_status ON seeds(seed_type, status);
-- 索引：按评分排序（用于种子调度）
CREATE INDEX idx_seeds_score ON seeds(score DESC);
```

### 2.3 原始证据 RawEvidence

> 采集器产出的一条原始结果。尚未经过 LLM 抽取。

```typescript
interface RawEvidence {
  id: string;
  sourceId: string;              // 来源 IntelSource ID
  seedIds: string[];             // 触发本次采集的种子 ID
  url: string;                   // 原始 URL（规范化后）
  title: string;                 // 标题（采集器提取）
  content: string;               // 正文纯文本
  contentHtml?: string;          // 原始 HTML（可选保存）
  summary?: string;              // 摘要（采集器生成或截取）
  author?: string;               // 作者
  publishedAt?: string;          // 发布时间（原始解析）
  collectedAt: string;           // 采集时间
  images: ImageRef[];            // 文章内图片
  metadata: EvidenceMetadata;    // 元数据
  hash: string;                  // SimHash（用于去重）
  status: EvidenceStatus;
  errorMessage?: string;         // 采集/解析失败原因
}

interface ImageRef {
  url: string;                   // 图片 URL
  alt?: string;                  // alt 文本
  position: number;              // 在文章中的位置（第几张）
  ocrText?: string;              // OCR 识别结果
  ocrConfidence?: number;        // OCR 置信度 0-1
  qwenDescription?: string;      // Qwen-VL 图片理解结果
  processed: boolean;            // 是否已处理
}

interface EvidenceMetadata {
  // 采集器信息
  collectorType: AccessMethod;   // 采集方式
  responseTime?: number;         // 响应耗时（ms）
  httpStatus?: number;           // HTTP 状态码
  // 页面信息
  language?: string;             // 页面语言
  wordCount?: number;            // 字数
  // RSS 特有
  guid?: string;                 // RSS GUID
  categories?: string[];         // RSS 分类标签
  // 搜索特有
  searchQuery?: string;          // 搜索关键词
  searchRank?: number;           // 搜索排名位置
}

type EvidenceStatus =
  | 'collected'      // 刚采集，待处理
  | 'extracting'     // LLM 抽取中
  | 'extracted'      // 抽取完成
  | 'failed'         // 处理失败
  | 'duplicate'      // 判定为重复
  | 'filtered';      // 被过滤（不满足条件）
```

**SQLite 表结构**：
```sql
CREATE TABLE raw_evidence (
  id TEXT PRIMARY KEY,
  source_id TEXT NOT NULL REFERENCES intel_sources(id),
  seed_ids TEXT NOT NULL DEFAULT '[]',       -- JSON
  url TEXT NOT NULL,
  title TEXT NOT NULL,
  content TEXT NOT NULL,
  content_html TEXT,
  summary TEXT,
  author TEXT,
  published_at TEXT,
  collected_at TEXT NOT NULL DEFAULT (datetime('now')),
  images TEXT NOT NULL DEFAULT '[]',         -- JSON
  metadata TEXT NOT NULL DEFAULT '{}',       -- JSON
  hash TEXT NOT NULL,
  status TEXT NOT NULL DEFAULT 'collected',
  error_message TEXT,
  UNIQUE(url)                                -- URL 去重
);

-- 索引：URL 规范化后查重
CREATE INDEX idx_evidence_url ON raw_evidence(url);
-- 索引：SimHash 查重
CREATE INDEX idx_evidence_hash ON raw_evidence(hash);
-- 索引：按状态查询待处理
CREATE INDEX idx_evidence_status ON raw_evidence(status);
-- 索引：按采集时间排序
CREATE INDEX idx_evidence_collected_at ON raw_evidence(collected_at DESC);
```

### 2.4 结构化事件 StructuredEvent

> LLM 从 RawEvidence 中抽取出的结构化情报。

```typescript
interface StructuredEvent {
  id: string;
  evidenceId: string;            // 来源 RawEvidence ID
  sourceId: string;              // 冗余，方便查询

  // —— 核心情报字段 ——
  eventTitle: string;            // 事件标题（一句话概括）
  keyFacts: KeyFact[];           // 关键事实
  actionAdvice: ActionAdvice[];  // 行动建议（按角色）
  eventType: EventType;          // 事件类型
  sentiment: Sentiment;          // 情绪倾向

  // —— 评估字段 ——
  impactScore: number;           // 影响评分 0-100
  priority: Priority;            // 优先级
  audienceTags: string[];        // 适用角色标签

  // —— 实体识别 ——
  entities: MentionedEntity[];   // 提及的实体

  // —— 元数据 ——
  extractedAt: string;           // 抽取时间
  model: string;                 // 使用的模型
  confidence: number;            // 抽取置信度 0-1
}

interface KeyFact {
  fact: string;                  // 事实描述
  importance: 'high' | 'medium' | 'low';
  entities: string[];            // 涉及的实体名
}

interface ActionAdvice {
  role: string;                  // 角色：'老板' | '发行' | '运营' ...
  advice: string;                // 建议内容
  urgency: 'immediate' | 'watch' | 'info';
}

interface Sentiment {
  polarity: 'positive' | 'negative' | 'neutral' | 'mixed';
  intensity: number;             // 强度 0-1
  target?: string;               // 情绪指向对象
}

interface MentionedEntity {
  name: string;                  // 实体名称
  type: 'game' | 'company' | 'person' | 'brand';
  role: 'subject' | 'object' | 'context';  // 在事件中的角色
  seedId?: string;               // 匹配到的种子 ID
}
```

**SQLite 表结构**：
```sql
CREATE TABLE structured_events (
  id TEXT PRIMARY KEY,
  evidence_id TEXT NOT NULL REFERENCES raw_evidence(id),
  source_id TEXT NOT NULL REFERENCES intel_sources(id),

  event_title TEXT NOT NULL,
  key_facts TEXT NOT NULL DEFAULT '[]',      -- JSON
  action_advice TEXT NOT NULL DEFAULT '[]',  -- JSON
  event_type TEXT NOT NULL,
  sentiment TEXT NOT NULL DEFAULT '{}',      -- JSON
  impact_score REAL NOT NULL DEFAULT 0,
  priority TEXT NOT NULL DEFAULT 'P2',
  audience_tags TEXT NOT NULL DEFAULT '[]',  -- JSON
  entities TEXT NOT NULL DEFAULT '[]',       -- JSON

  extracted_at TEXT NOT NULL DEFAULT (datetime('now')),
  model TEXT NOT NULL,
  confidence REAL NOT NULL DEFAULT 0
);

-- 索引：按事件类型查询
CREATE INDEX idx_events_type ON structured_events(event_type);
-- 索引：按评分排序
CREATE INDEX idx_events_score ON structured_events(impact_score DESC);
-- 索引：按来源证据查询
CREATE INDEX idx_events_evidence ON structured_events(evidence_id);
```

### 2.5 证据事件 EvidenceEvent

> 多个 StructuredEvent 经去重合并后的统一事件。同一事件在不同源出现会合并到这里。

```typescript
interface EvidenceEvent {
  id: string;
  
  // —— 合并后的事件信息 ——
  eventTitle: string;            // 最佳标题（选最完整/最清晰的）
  eventType: EventType;
  keyFacts: KeyFact[];           // 合并去重后的关键事实
  actionAdvice: ActionAdvice[];  // 合并去重后的行动建议
  sentiment: Sentiment;          // 综合情绪
  
  // —— 来源与可信度 ——
  evidenceIds: string[];         // 合并的 RawEvidence ID 列表
  structuredEventIds: string[];  // 合并的 StructuredEvent ID 列表
  sourceCount: number;           // 报道此事件的源数量
  sourceIds: string[];           // 报道此事件的源 ID 列表
  
  // —— 评估 ——
  impactScore: number;           // 综合影响评分
  confidenceScore: number;       // 可信度评分（多源交叉验证）
  priority: Priority;
  audienceTags: string[];
  
  // —— 实体 ——
  entities: MentionedEntity[];   // 合并后的实体列表
  relatedSeedIds: string[];      // 相关种子
  
  // —— 时间线 ——
  firstSeenAt: string;           // 首次发现时间
  lastSeenAt: string;            // 最近发现时间（新报道）
  publishedAt?: string;          // 事件发生时间（LLM 推断）
  createdAt: string;
  updatedAt: string;
  
  // —— 去重记录 ——

}
```

### 2.6 信号 Signal

> 经过评分的、可消费的情报单元。推送给前端 / Dataki / 日报。

```typescript
interface Signal {
  id: string;
  evidenceEventId: string;       // 关联的 EvidenceEvent
  sourceId: string;
  
  // —— 信号内容 ——
  title: string;                 // 信号标题（面向用户）
  summary: string;               // 一段话摘要
  eventType: EventType;
  priority: Priority;
  impactScore: number;
  
  // —— 分类标签 ——
  audienceTags: string[];        // 适用角色
  topicTags: string[];           // 话题标签
  entityTags: string[];          // 实体标签（游戏/公司）
  
  // —— 信号状态 ——
  status: SignalStatus;
  readByRoles: string[];         // 已通知的角色
  dispatchedAt?: string;         // 推送时间
  createdAt: string;
}

type SignalStatus =
  | 'new'          // 新信号
  | 'dispatched'   // 已推送
  | 'consumed'     // 已消费
  | 'archived';    // 已归档
```

**SQLite 表结构**：
```sql
CREATE TABLE signals (

  summary TEXT NOT NULL,
  event_type TEXT NOT NULL,
  priority TEXT NOT NULL,
  impact_score REAL NOT NULL,
  
  audience_tags TEXT NOT NULL DEFAULT '[]',
```

## 四、设计决策记录

| 决策 | 结论 | 理由 |
|------|------|------|
| 数据库选型 | SQLite | 零依赖、查询方便、单机足够 |
| JSON 字段存储 | SQLite TEXT + JSON 序列化/反序列化 | 灵活结构用 JSON，查询条件用独立列 |
| RawEvidence 保存 HTML | 可选，默认不保存 | 节省空间，调试时可按需开启 |
| 去重策略 | SimHash + URL 规范化 | 轻量级，语义检索走 Dataki |
| 种子与证据关系 | 多对多（seed_ids 数组） | 一条证据可能由多个种子联合触发 |
| 事件类型用中文 | ✅ | 保持业务语义，符合 CLAUDE.md 规范 |
| 时间格式 | ISO 8601 字符串 | SQLite 友好，前后端统一 |
| 图片存储 | 只存 URL + OCR 结果 | 不下载保存图片二进制 |
